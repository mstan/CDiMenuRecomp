cmake_minimum_required(VERSION 3.20)
project(CDiMenuRecomp VERSION 0.2.0 LANGUAGES C CXX)

set(BUILD_TESTING OFF)
set(CDI_BUILD_RUNTIME ON)
set(CDI_COSIM_BUILD OFF CACHE BOOL "" FORCE)
set(CDI_GAME_GENERATED_DIR "")
set(CDI_RUNTIME_TARGET CdiMenu)
set(CDI_DEFER_PROVENANCE ON)
set(RECOMP_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/recomp-ui")
set(RECOMP_UI_SDL3 OFF CACHE BOOL "" FORCE)
foreach(component VIDEO LOADER DRAWING MEMORY SERVICE STREAMING BOARD GPU)
    set(CDI_${component}_BACKEND HLE CACHE STRING "" FORCE)
endforeach()

find_package(SDL2 CONFIG REQUIRED)
add_executable(CdiMenu src/launcher.c)
add_subdirectory(cdirecomp/runner cdi-runtime)
include("${RECOMP_UI_ROOT}/recomp_ui.cmake")
recomp_target_launcher_ui(CdiMenu CONSOLE cdi)
target_compile_definitions(CdiMenu PRIVATE CDI_PRODUCT_NAME="CD-i Menu" CDI_PRODUCT_VERSION="${PROJECT_VERSION}")
set_target_properties(CdiMenu PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

include("${CMAKE_CURRENT_SOURCE_DIR}/cdirecomp/cmake/CdiProvenance.cmake")
cdirecomp_finalize_player(CdiMenu)
